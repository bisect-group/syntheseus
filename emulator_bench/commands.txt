# Run all commands from the syntheseus submodule root: /home/adhil/github/EMULaToR/baselines/syntheseus
# Environment for the full run: conda env "syntheseus_modern".
# The pinned env "syntheseus" (environment.yml) is kept as a reference only.
PY=/home/adhil/.conda/envs/syntheseus_modern/bin/python
OUT=/home/adhil/github/EMULaToR/outputs/baselines/syntheseus

# 1. Corpora (downloads, manifest in $OUT/corpora/manifest.json)
$PY -m emulator_bench.corpora

# 2. Checkpoints. The syntheseus default checkpoints download on first use into
#    $OUT/checkpoints/syntheseus_default. The two forward checkpoints come from the authors:
#    Chemformer: fined-tuned/uspto_mixed/last.ckpt -> $OUT/checkpoints/Chemformer/forward/original/last.ckpt
#    R-SMILES:   models/RtoP/USPTO-MIT_RtoP_mixed.pt -> $OUT/checkpoints/RootAligned/forward/
$PY -m emulator_bench.prepare_checkpoints

# 3. GLN checkout and C++ library (no CUDA build needed)
git clone https://github.com/kmaziarz/GLN $OUT/external/GLN
git -C $OUT/external/GLN checkout 5437fa7a7cfe98354aa5b0e743b964a97971016c
make -C $OUT/external/GLN/gln/mods/mol_gnn/mg_clib

# 3b. Curated retro split files (see README "Retro targets"). The mapper needs rxnmapper (any env).
$PY -m emulator_bench.retro_targets prepare
<python with rxnmapper> -I emulator_bench/map_reactions.py <requests.parquet> <maps.parquet>   # paths printed by prepare
$PY -m emulator_bench.retro_targets build

# 4. Overlap audit (CPU, 16 threads): writes $OUT/overlap/ and $OUT/summary_overlap_by_corpus.csv
taskset -c 0-15 $PY -u -m emulator_bench.overlap

# 5. Dry run, then the full run (every model x direction x split folder; test fold only)
$PY -m emulator_bench.launch --gpus 0 --workers 4 --procs_per_gpu 4 --dry_run
$PY -m emulator_bench.launch --gpus 0 --workers 4 --procs_per_gpu 4
# Determinism: predict.py caps the process pools at 8 workers (RootAligned retro and MHNreact fork cpu_count() workers
# per call). build_model keeps model.eval() for LocalRetro and RetroKNN and seeds RootAligned retro per input (CRC32).
# launch.py sets PYTHONHASHSEED=0 for every child process.

# 6. One model, one split (--folds train val test also works)
$PY -m emulator_bench.launch --gpus 0 --models LocalRetro --directions retro \
    --splits random_splits_grouped_rxn_key --folds test

# 6b. Seeds 1 and 2 (seed 0 is the default run): same commands with --seed, runs go to $OUT/seed<N>
$PY -m emulator_bench.launch --directions retro --gpus 3 --seed 1
$PY -m emulator_bench.launch --directions retro --gpus 3 --seed 2

# 7. Summaries: common test set, retro variants, then $OUT/summary_topk.csv, summary_topk_seeds.csv,
#    summary_overlap.csv
$PY -m emulator_bench.common_set --seeds 0 1 2
$PY -m emulator_bench.rescore
$PY -m emulator_bench.aggregate

# Smoke test (32 rows)
S=<scratch dir>
$PY -m emulator_bench.launch --runs_root $S --gpus 0 --workers 2 --procs_per_gpu 2 \
    --splits random_splits_grouped_rxn_key --folds test --limit 32

# Environment comparison (ranked top-10 lists and max score difference of two run folders)
$PY -m emulator_bench.compare_runs $S/pinned/retro/LocalRetro/random_splits_grouped_rxn_key/all/test \
    $S/modern/retro/LocalRetro/random_splits_grouped_rxn_key/all/test

# Equivalence with the syntheseus CLI (retro, same JSONL)
syntheseus eval-single-step data_dir=<jsonl dir with test.jsonl> model_class=LocalRetro fold=TEST \
    num_top_results=50 results_dir=<dir> print_idxs=[1,10,50] include_predictions=True

# Checks
$PY -m pytest emulator_bench/tests -q
ruff check emulator_bench && ruff format --check emulator_bench
git diff --check

# ---------------------------------------------------------------------------
# Fine-tuning (see README "Fine-tuning"). Needs the curated retro files first.
# ---------------------------------------------------------------------------

# 1. Per-model training data. LocalRetro's CSVs also feed MEGAN; RootAligned has its own variant
#    because it needs every product atom mapped.
$PY -m emulator_bench.finetune_data --model LocalRetro
$PY -m emulator_bench.finetune_data --model RootAligned
$PY -m emulator_bench.finetune_data --model Chemformer --skip_maps

# 2. Train one cell (model x direction x split x seed). Seed 0 keeps the original paths.
$PY -m emulator_bench.finetune_chemformer  --split random_splits_grouped_rxn_key --direction retro --seed 0
$PY -m emulator_bench.finetune_rootaligned --split random_splits_grouped_rxn_key --direction forward --seed 0
$PY -m emulator_bench.finetune_megan       --split random_splits_grouped_rxn_key --seed 0   # --direction retro
$PY -m emulator_bench.finetune_megan       --split random_splits_grouped_rxn_key --seed 0 --direction forward
$PY -m emulator_bench.finetune_localretro  --split random_splits_grouped_rxn_key            # retro only

# 3. Evaluate it through the usual path (model, group, name, shards, direction, seed)
$OUT/finetune_eval.sh Chemformer random_splits_grouped_rxn_key all 6 retro 0

# 4. The whole matrix across GPUs: one worker per GPU, each claiming cells atomically. Resumable.
$OUT/parallel_worker.sh 0 &
$OUT/parallel_worker.sh 1 &
$OUT/parallel_worker.sh 2 &

# 4b. Forward MEGAN, which parallel_worker.sh does not cover. Unlike finetune_eval.sh (which pins
#     CUDA_VISIBLE_DEVICES=0) this takes the GPU, so training and evaluation stay on it.
#     usage: megan_forward_run.sh <split_key> <seed> <gpu> [shards]
$OUT/megan_forward_run.sh random_splits_grouped_rxn_key 0 2 4

# 5. USPTO-50K control: the same wrapper on the data the checkpoints were trained on, which is how
#    the zero-shot numbers were shown to be a property of the data and not of this code.
$PY -m emulator_bench.checks.uspto50k_control --models LocalRetro --limit 500
